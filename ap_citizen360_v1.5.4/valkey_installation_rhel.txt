# Option A: Install from source (RHEL 8/9)
sudo dnf groupinstall "Development Tools" -y
sudo dnf install -y openssl-devel

wget https://github.com/valkey-io/valkey/archive/refs/tags/8.1.0.tar.gz
tar -xzf 8.1.0.tar.gz
cd valkey-8.1.0
make -j$(nproc)
sudo make install

# Start Valkey as a background service
valkey-server --daemonize yes --port 6379 --bind 127.0.0.1

# Or create a systemd service (recommended for production)
sudo tee /etc/systemd/system/valkey.service <<EOF
[Unit]
Description=Valkey In-Memory Data Store
After=network.target

[Service]
ExecStart=/usr/local/bin/valkey-server --port 6379 --bind 127.0.0.1
Restart=always
User=nobody

[Install]
WantedBy=multi-user.target
EOF

sudo systemctl daemon-reload
sudo systemctl enable --now valkey
